\begin{frame}{Mô hình dữ liệu chính}
\begin{itemize}
\setlength{\itemsep}{8pt}
\item \textbf{Tổ chức:} User thuộc Department; Project có phòng ban sở hữu và tham gia.
\item \textbf{Công việc:} Project chứa WorkRequest; mỗi WorkRequest có nhiều Task.
\item \textbf{Thời gian:} Attendance ghi nhận theo ngày, AttendanceSession lưu từng phiên vào/ra.
\item \textbf{Phê duyệt:} LeaveRequest, WorkLocationRequest và AttendanceAdjustmentRequest lưu trạng thái xử lý.
\item \textbf{Chốt số liệu:} PayrollPeriod biểu diễn kỳ công theo năm và tháng.
\end{itemize}
\end{frame}

\begin{frame}{Quy trình dự án và task}
\begin{enumerate}
\setlength{\itemsep}{10pt}
\item Tạo dự án và xác định các phòng ban tham gia.
\item Tạo work request với mục tiêu, người điều phối và hạn xử lý.
\item Chia yêu cầu thành task, gán người thực hiện và người kiểm thử.
\item Cập nhật trạng thái theo luồng hợp lệ: Backlog $\rightarrow$ Todo $\rightarrow$ In progress $\rightarrow$ Review $\rightarrow$ Testing $\rightarrow$ Done.
\end{enumerate}
\vspace{5mm}
\small Task chuyển sang Blocked phải có lý do; work request không được hoàn tất khi còn task mở.
\end{frame}

\begin{columns}[T,onlytextwidth]

\textbf{Ghi nhận công}

\begin{itemize}

\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Nghỉ phép và địa điểm làm việc}
\begin{itemize}
\setlength{\itemsep}{10pt}
\item Đơn nghỉ lưu loại nghỉ, khoảng ngày, phần ngày và lý do.
\item Bộ tính ngày làm việc xử lý cả trường hợp nghỉ nửa ngày.
\item Đơn nghỉ năm liên quan đến hạn mức và số dư phép.
\item Yêu cầu làm việc tại nhà hoặc tại khách hàng được quản lý riêng với nghỉ phép.
\item Người gửi không tự phê duyệt yêu cầu của mình.
\end{itemize}
\end{frame}

\begin{frame}{Bảng công tháng}
\begin{itemize}
\setlength{\itemsep}{10pt}
\item Tổng hợp dữ liệu chấm công và các đơn nghỉ đã được duyệt theo kỳ.
\item Trình bày ngày làm việc, thời gian làm, đi muộn và các loại nghỉ.
\item Xuất tệp XLSX để đối soát.
\item Khóa kỳ công sau khi chốt; mở lại kỳ cần có lý do.
\end{itemize}
\vspace{5mm}
\small Đây là báo cáo ngày công, chưa phải bảng tính lương thực nhận.
\end{frame}

\begin{frame}{Xác thực và phân quyền}
\begin{itemize}
\setlength{\itemsep}{10pt}
\item Người dùng đăng nhập và nhận JWT; API nghiệp vụ yêu cầu token hợp lệ.
\item Mật khẩu được kiểm tra qua BCrypt.
\item Quyền được xét theo vai trò và phạm vi bản ghi: cá nhân, phòng ban hoặc dự án.
\item Service kiểm tra thêm trạng thái nghiệp vụ trước khi cập nhật.
\end{itemize}
\end{frame}
